function(chord_target target midi formats code suffix)
 if(midi)
  set(synth FALSE)
 else()
  set(synth TRUE)
 endif()
 juce_add_plugin(${target} COMPANY_NAME "KHIMAIRA" BUNDLE_ID "${bundle}.${suffix}" VERSION ${PROJECT_VERSION}
  IS_SYNTH ${synth} IS_MIDI_EFFECT ${midi} NEEDS_MIDI_INPUT TRUE NEEDS_MIDI_OUTPUT TRUE
  EDITOR_WANTS_KEYBOARD_FOCUS TRUE COPY_PLUGIN_AFTER_BUILD FALSE
  PLUGIN_MANUFACTURER_CODE Khmr PLUGIN_CODE ${code}
  VST3_CATEGORIES Instrument Generator AU_MAIN_TYPE kAudioUnitType_MIDIProcessor
  FORMATS ${formats} PRODUCT_NAME "${brand} 1.5 Preview${suffixName}")
 target_sources(${target} PRIVATE Source/PluginProcessor.cpp Source/PluginEditor.cpp)
 target_compile_definitions(${target} PUBLIC JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_VST3_CAN_REPLACE_VST2=0 JUCE_DISPLAY_SPLASH_SCREEN=0 KHIMAIRA_MIDI_FX=$<BOOL:${midi}> CHORD_SEIREN=${isSeiren})
 target_link_libraries(${target} PRIVATE ChordResources juce::juce_javascript juce::juce_audio_utils juce::juce_gui_extra PUBLIC juce::juce_recommended_config_flags)
endfunction()
